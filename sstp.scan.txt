#!/data/data/com.termux/files/usr/bin/bash

URL='http://150.40.105.8:61446/'

GREEN='\033[1;32m'
RED='\033[1;31m'
YELLOW='\033[1;33m'
CYAN='\033[1;36m'
WHITE='\033[1;37m'
RESET='\033[0m'

TMP="$HOME/sstp_list.html"

clear

echo "=============================================="
echo "          @vpn_proxy66"
echo "          SSTP SERVER SCANNER"
echo "=============================================="
echo

for CMD in curl grep sed awk sort timeout; do
    if ! command -v "$CMD" >/dev/null 2>&1; then
        echo -e "${RED}[!] Missing: $CMD${RESET}"
        echo
        echo "Run:"
        echo "pkg update"
        echo "pkg install curl grep sed gawk coreutils"
        exit 1
    fi
done

echo -e "${CYAN}[*] Downloading server list...${RESET}"

if ! curl \
    -L \
    --silent \
    --show-error \
    --fail \
    --connect-timeout 15 \
    --max-time 40 \
    -A "Mozilla/5.0" \
    "$URL" \
    -o "$TMP"
then
    echo
    echo -e "${RED}[!] Failed to download server list.${RESET}"
    rm -f "$TMP"
    read -p "Press Enter..."
    exit 1
fi

if [ ! -s "$TMP" ]; then
    echo -e "${RED}[!] Empty server list.${RESET}"
    rm -f "$TMP"
    read -p "Press Enter..."
    exit 1
fi

echo -e "${GREEN}[+] Server list downloaded.${RESET}"
echo

mapfile -t SERVERS < <(
    grep -Eo \
    '[A-Za-z0-9.-]+\.opengw\.net(:[0-9]{1,5})?' \
    "$TMP" |
    sort -u
)

if [ "${#SERVERS[@]}" -eq 0 ]; then
    echo
    echo -e "${RED}[!] No SSTP servers found.${RESET}"
    rm -f "$TMP"
    read -p "Press Enter..."
    exit 1
fi

echo -e "${GREEN}[+] Found ${#SERVERS[@]} servers.${RESET}"
echo

ONLINE=0
TOTAL=0

for SERVER in "${SERVERS[@]}"; do

    TOTAL=$((TOTAL + 1))

    if [[ "$SERVER" == *:* ]]; then
        HOST="${SERVER%:*}"
        PORT="${SERVER##*:}"
    else
        HOST="$SERVER"
        PORT="443"
    fi

    IP=""

    if command -v getent >/dev/null 2>&1; then
        IP=$(getent hosts "$HOST" 2>/dev/null |
             awk 'NR==1 {print $1}')
    fi

    if [ -z "$IP" ]; then
        IP=$(ping -c 1 -W 2 "$HOST" 2>/dev/null |
             grep -Eo '([0-9]{1,3}\.){3}[0-9]{1,3}' |
             head -1)
    fi

    # ------------------------------------------
    # ONLINE
    # ------------------------------------------

    if [ -n "$IP" ] && timeout 5 bash -c \
        "echo >/dev/tcp/$HOST/$PORT" \
        >/dev/null 2>&1
    then

        ONLINE=$((ONLINE + 1))

        echo -e "${GREEN}╔══════════════════════════════════════╗${RESET}"
        echo -e "${GREEN}║${RESET}            ${CYAN}@vpn_proxy66${RESET}             ${GREEN}║${RESET}"
        echo -e "${GREEN}╠══════════════════════════════════════╣${RESET}"
        echo -e "${GREEN}║${RESET} ${WHITE}STATUS${RESET}   : ${GREEN}ONLINE${RESET}              ${GREEN}║${RESET}"
        echo -e "${GREEN}║${RESET} ${WHITE}SERVER${RESET}   : $HOST"
        echo -e "${GREEN}║${RESET} ${WHITE}PORT${RESET}     : $PORT"
        echo -e "${GREEN}║${RESET} ${WHITE}IP${RESET}       : $IP"
        echo -e "${GREEN}║${RESET} ${WHITE}USERNAME${RESET} : vpn"
        echo -e "${GREEN}║${RESET} ${WHITE}PASSWORD${RESET} : vpn"
        echo -e "${GREEN}╚══════════════════════════════════════╝${RESET}"
        echo

    fi

done

rm -f "$TMP"

echo "=============================================="
echo -e "          ${CYAN}@vpn_proxy66${RESET}"
echo "=============================================="
echo
echo -e "Total servers : ${CYAN}$TOTAL${RESET}"
echo -e "Online        : ${GREEN}$ONLINE${RESET}"
echo
echo "=============================================="

read -p "Press Enter..."
