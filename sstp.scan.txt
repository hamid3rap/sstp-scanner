#                 @vpn_proxy66
#          SSTP + L2TP/IPsec SCANNER
# ==========================================================

SOURCE="http://150.40.105.5:32536/"
TMP="$HOME/vpn_source.html"

GREEN='\033[1;32m'
RED='\033[1;31m'
YELLOW='\033[1;33m'
CYAN='\033[1;36m'
WHITE='\033[1;37m'
RESET='\033[0m'

# ==========================================================
# REQUIREMENTS
# ==========================================================

check_requirements() {

    for CMD in curl grep sed awk sort ping timeout; do
        if ! command -v "$CMD" >/dev/null 2>&1; then
            echo -e "${RED}[!] Missing: $CMD${RESET}"
            echo
            echo "Install with:"
            echo
            echo "pkg update"
            echo "pkg install curl grep sed gawk coreutils"
            exit 1
        fi
    done
}

# ==========================================================
# HEADER
# ==========================================================

header() {

    clear

    echo "=============================================="
    echo -e "              ${CYAN}@vpn_proxy66${RESET}"
    echo "=============================================="
    echo
}

# ==========================================================
# DOWNLOAD SOURCE
# ==========================================================

download_source() {

    rm -f "$TMP"

    echo -e "${CYAN}[*] Downloading server list...${RESET}"

    if ! curl \
        -L \
        --silent \
        --show-error \
        --fail \
        --connect-timeout 15 \
        --max-time 40 \
        -A "Mozilla/5.0" \
        "$SOURCE" \
        -o "$TMP"
    then
        echo
        echo -e "${RED}[!] Failed to download source.${RESET}"
        return 1
    fi

    if [ ! -s "$TMP" ]; then
        echo
        echo -e "${RED}[!] Source is empty.${RESET}"
        return 1
    fi

    echo -e "${GREEN}[+] Source downloaded.${RESET}"
    echo

    return 0
}

# ==========================================================
# SSTP RESULT BOX
# ==========================================================

show_sstp() {

    HOST="$1"
    PORT="$2"
    IP="$3"

    echo -e "${GREEN}╔══════════════════════════════════════════╗${RESET}"
    echo -e "${GREEN}║${RESET}              ${CYAN}@vpn_proxy66${RESET}              ${GREEN}║${RESET}"
    echo -e "${GREEN}╠══════════════════════════════════════════╣${RESET}"
    echo -e "${GREEN}║${RESET} TYPE     : SSTP"
    echo -e "${GREEN}║${RESET} STATUS   : ${GREEN}ONLINE${RESET}"
    echo -e "${GREEN}║${RESET} SERVER   : $HOST"
    echo -e "${GREEN}║${RESET} PORT     : $PORT"
    echo -e "${GREEN}║${RESET} IP       : $IP"
    echo -e "${GREEN}║${RESET} USERNAME : vpn"
    echo -e "${GREEN}║${RESET} PASSWORD : vpn"
    echo -e "${GREEN}╚══════════════════════════════════════════╝${RESET}"
    echo
}

# ==========================================================
# L2TP RESULT BOX
# ==========================================================

show_l2tp() {

    IP="$1"

    echo -e "${GREEN}╔══════════════════════════════════════════╗${RESET}"
    echo -e "${GREEN}║${RESET}              ${CYAN}@vpn_proxy66${RESET}              ${GREEN}║${RESET}"
    echo -e "${GREEN}╠══════════════════════════════════════════╣${RESET}"
    echo -e "${GREEN}║${RESET} TYPE     : L2TP/IPsec"
    echo -e "${GREEN}║${RESET} STATUS   : ${GREEN}ONLINE${RESET}"
    echo -e "${GREEN}║${RESET} SERVER IP: $IP"
    echo -e "${GREEN}║${RESET} PSK      : vpn"
    echo -e "${GREEN}║${RESET} USERNAME : vpn"
    echo -e "${GREEN}║${RESET} PASSWORD : vpn"
    echo -e "${GREEN}╚══════════════════════════════════════════╝${RESET}"
    echo
}

# ==========================================================
# SSTP SCANNER
# ==========================================================

scan_sstp() {

    header

    echo -e "${CYAN}              SSTP SCANNER${RESET}"
    echo
    echo "Source: $SOURCE"
    echo

    if ! download_source; then
        read -p "Press Enter..."
        return
    fi

    echo -e "${CYAN}[*] Finding SSTP records...${RESET}"
    echo

    TOTAL=0
    ONLINE=0

    # Each </tr> is treated as one VPN server record.
    while IFS= read -r ROW; do

        # Only process records that actually contain SSTP
        if ! grep -qi "howto_sstp.aspx" <<< "$ROW"; then
            continue
        fi

        TOTAL=$((TOTAL + 1))

        # Extract SSTP hostname from the SSTP part of this record
        HOSTPORT=$(echo "$ROW" |
            grep -Eo \
            'SSTP Hostname[[:space:]]*:?[[:space:]]*[^<" ]+\.opengw\.net(:[0-9]+)?' |
            grep -Eo \
            '[A-Za-z0-9.-]+\.opengw\.net(:[0-9]+)?' |
            head -1)

        # Fallback if text is encoded differently
        if [ -z "$HOSTPORT" ]; then
            HOSTPORT=$(echo "$ROW" |
                grep -Eo \
                '[A-Za-z0-9.-]+\.opengw\.net(:[0-9]{2,5})?' |
                head -1)
        fi

        [ -z "$HOSTPORT" ] && continue

        if [[ "$HOSTPORT" == *:* ]]; then
            HOST="${HOSTPORT%:*}"
            PORT="${HOSTPORT##*:}"
        else
            HOST="$HOSTPORT"
            PORT="443"
        fi

        # Resolve hostname
        IP=""

        if command -v getent >/dev/null 2>&1; then
            IP=$(getent hosts "$HOST" 2>/dev/null |
                awk 'NR==1 {print $1}')
        fi

        if [ -z "$IP" ]; then
            IP=$(ping -c 1 -W 2 "$HOST" 2>/dev/null |
                grep -Eo '([0-9]{1,3}\.){3}[0-9]{1,3}' |
                head -1)
        fi

        # TCP test
        if [ -n "$IP" ] &&
            timeout 5 bash -c \
            "echo >/dev/tcp/$HOST/$PORT" \
            >/dev/null 2>&1
        then

            ONLINE=$((ONLINE + 1))

            show_sstp "$HOST" "$PORT" "$IP"

        fi

    done < <(
        awk 'BEGIN { RS="</tr>" } { print }' "$TMP"
    )

    echo "=============================================="
    echo -e "              ${CYAN}@vpn_proxy66${RESET}"
    echo "=============================================="
    echo
    echo "SSTP records : $TOTAL"
    echo -e "Online       : ${GREEN}$ONLINE${RESET}"
    echo

    read -p "Press Enter..."
}

# ==========================================================
# L2TP/IPsec SCANNER
# ==========================================================

scan_l2tp() {

    header

    echo -e "${CYAN}             L2TP/IPsec SCANNER${RESET}"
    echo
    echo "Source: $SOURCE"
    echo

    if ! download_source; then
        read -p "Press Enter..."
        return
    fi

    echo -e "${CYAN}[*] Finding L2TP/IPsec records...${RESET}"
    echo

    TOTAL=0
    ONLINE=0

    # IMPORTANT:
    # We only process records containing:
    # howto_l2tp.aspx
    #
    # Therefore random IP addresses from other VPN
    # protocols will NOT be treated as L2TP.

    while IFS= read -r ROW; do

        if ! grep -qi "howto_l2tp.aspx" <<< "$ROW"; then
            continue
        fi

        TOTAL=$((TOTAL + 1))

        # Extract IPv4 from THIS SAME SERVER RECORD
        IP=$(echo "$ROW" |
            grep -Eo \
            '([0-9]{1,3}\.){3}[0-9]{1,3}' |
            awk -F. '
            $1<=255 &&
            $2<=255 &&
            $3<=255 &&
            $4<=255 {
                print
            }' |
            head -1)

        [ -z "$IP" ] && continue

        # ------------------------------------------
        # Basic reachability
        # ------------------------------------------

        if ping -c 1 -W 2 "$IP" >/dev/null 2>&1; then

            ONLINE=$((ONLINE + 1))

            show_l2tp "$IP"

        fi

    done < <(
        awk 'BEGIN { RS="</tr>" } { print }' "$TMP"
    )

    echo "=============================================="
    echo -e "              ${CYAN}@vpn_proxy66${RESET}"
    echo "=============================================="
    echo
    echo "L2TP records : $TOTAL"
    echo -e "Reachable    : ${GREEN}$ONLINE${RESET}"
    echo

    read -p "Press Enter..."
}

# ==========================================================
# MAIN MENU
# ==========================================================

check_requirements

while true; do

    header

    echo "                 VPN SCANNER"
    echo
    echo -e "${GREEN}[1]${RESET} SSTP"
    echo -e "${GREEN}[2]${RESET} L2TP/IPsec"
    echo -e "${RED}[0]${RESET} Exit"
    echo
    echo "=============================================="
    echo

    read -p "Select option: " OPTION

    case "$OPTION" in

        1)
            scan_sstp
            ;;

        2)
            scan_l2tp
            ;;

        0)
            rm -f "$TMP"
            clear
            exit 0
            ;;

        *)
            echo
            echo -e "${RED}[!] Invalid option.${RESET}"
            sleep 1
            ;;

    esac

done
~ $
