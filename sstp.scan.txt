#!/data/data/com.termux/files/usr/bin/bash

RED="\033[31m"
GREEN="\033[32m"
ORANGE="\033[33m"
CYAN="\033[36m"
MAGENTA="\033[35m"
RESET="\033[0m"
YELLOW="\033[1;33m"

TAG="${MAGENTA}┌────────────────────┐
│ t.me/ShadowProxy66 │
└────────────────────┘${RESET}"

# ----------------------------------------------------------
# GeoIP Country Lookup
# ----------------------------------------------------------
get_country() {
    ip="$1"
    curl -fsSL "http://ip-api.com/json/$ip" | jq -r '.country // "Unknown"'
}

# ----------------------------------------------------------
# Option 1 — SSTP Scanner
# ----------------------------------------------------------
scan_sstp() {
    PAGE_URL="https://ipspeed.info/free-sstp.php"

    echo "[*] Fetching SSTP servers..."
    HTML="$(curl -fsSL "$PAGE_URL")" || { echo "[!] Fetch failed"; return; }

    mapfile -t RAW_HOSTS < <(echo "$HTML" | grep -Eo '[a-zA-Z0-9.-]+\.[a-z]{2,6}:[0-9]{2,5}')

    if [[ ${#RAW_HOSTS[@]} -eq 0 ]]; then
      echo "[!] No hosts found"
      return
    fi

    echo "[*] Testing servers..."
    for item in "${RAW_HOSTS[@]}"; do
      host="${item%%:*}"
      port="${item##*:}"
      [[ "$host" == "$port" ]] && port=443

      country=$(get_country "$host")

      if timeout 5 bash -c "echo > /dev/tcp/${host}/${port}" 2>/dev/null; then
        echo -e "${GREEN}[OPEN] ${host}:${port} (${country})${RESET}\n$TAG"
      else
        echo -e "${RED}[CLOSED] ${host}:${port} (${country})${RESET}\n$TAG"
      fi
    done
}

# ----------------------------------------------------------
# Option 2 — DoH Scanner
# ----------------------------------------------------------
scan_doh() {

PAGE_URL="https://github.com/curl/curl/wiki/DNS-over-HTTPS#publicly-available-servers"
RESULTS="$HOME/.cache/doh_results.txt"

mkdir -p "$(dirname "$RESULTS")"
rm -f "$RESULTS"

echo "[+] Fetching DoH list..."
RAW=$(curl -sL "$PAGE_URL")

DNS_LIST=($(printf '%s\n' "$RAW" | grep -oE 'https://[^ \"'\''()]+dns-query' | sort -u))
TOTAL=${#DNS_LIST[@]}

if [ "$TOTAL" -eq 0 ]; then
    echo "[!] No DoH servers found!"
    return
fi

echo "[+] Running parallel latency tests..."
BEST_DNS=""
BEST_TIME=999999
COUNT=0

for DNS in "${DNS_LIST[@]}"; do
(
    START=$(date +%s%3N)
    curl -s --http2 --max-time 2 -H 'accept: application/dns-json' "${DNS}?name=example.com&type=A" > /dev/null
    END=$(date +%s%3N)
    LATENCY=$((END - START))

    printf "%s %s\n" "$DNS" "$LATENCY" >> "$RESULTS"
) &
done

while [ "$COUNT" -lt "$TOTAL" ]; do
    COUNT=$(wc -l < "$RESULTS" 2>/dev/null)
    PERCENT=$((COUNT * 100 / TOTAL))
    echo -ne "🔄 Progress: ${PERCENT}% \r"
    sleep 0.1
done

echo ""

while read -r DNS LAT; do
    if [ "$LAT" -lt "$BEST_TIME" ] && [ "$LAT" -gt 0 ]; then
        BEST_TIME=$LAT
        BEST_DNS=$DNS
    fi
done < "$RESULTS"

echo "=============================="
echo -e "🔥 Fastest DNS-over-HTTPS server:"
echo -e "➡️ ${GREEN}$BEST_DNS${RESET}"
echo -e "⏱ Latency: ${GREEN}$BEST_TIME ms${RESET}"
echo "=============================="

rm -f "$RESULTS"
}

# ----------------------------------------------------------
# Option 3 — L2TP Scanner
# ----------------------------------------------------------
scan_l2tp() {

URL="https://sstp.mehdi-hoore.workers.dev/"

echo "=============================================="
echo "           L2TP/IPsec SERVER CHECKER"
echo "=============================================="
echo

echo -e "${YELLOW}[*] Downloading server list...${RESET}"

HTML=$(curl -fsSL --max-time 20 "$URL" 2>/dev/null)

if [ -z "$HTML" ]; then
    echo -e "${RED}[!] Failed to download server list.${RESET}"
    return
fi

echo -e "${GREEN}[+] Download successful.${RESET}"
echo

ROWS=$(printf '%s' "$HTML" |
    sed 's/<tr/\n<tr/g' |
    sed 's#</tr>#</tr>\n#g')

FOUND=0

echo -e "${CYAN}[*] Searching for L2TP/IPsec servers...${RESET}"
echo

while IFS= read -r ROW; do

    [ -z "$ROW" ] && continue

    if ! printf '%s' "$ROW" | grep -qi 'howto_l2tp\.aspx'; then
        continue
    fi

    HOST=$(printf '%s' "$ROW" |
        grep -Eo '(vpn[0-9]+|public-vpn-[0-9]+)\.opengw\.net' |
        head -n 1)

    IP=$(printf '%s' "$ROW" |
        grep -Eo '([0-9]{1,3}\.){3}[0-9]{1,3}' |
        head -n 1)

    if [ -z "$HOST" ] || [ -z "$IP" ]; then
        continue
    fi

    FOUND=$((FOUND + 1))

    echo -e "${YELLOW}Creator @vpn_proxy66${RESET}"
    echo -e "${CYAN}L2TP SERVER${RESET}"
    echo "Host : $HOST"
    echo "IP   : $IP"

    PING=$(ping -c 1 -W 2 "$IP" 2>/dev/null |
        grep -oE 'time=[0-9.]+ ?ms' |
        head -n 1)

    if [ -n "$PING" ]; then
        echo -e "Ping : ${GREEN}$PING${RESET}"
        echo -e "Status: ${GREEN}ONLINE${RESET}"
    else
        echo -e "Ping : ${RED}TIMEOUT${RESET}"
        echo -e "Status: ${RED}OFFLINE${RESET}"
    fi

    echo -e "${CYAN}Username:${RESET} vpn"
    echo -e "${CYAN}Password:${RESET} vpn"
    echo -e "${CYAN}IPsec Key:${RESET} vpn"

    echo "----------------------------------------------"

done <<< "$ROWS"

echo
echo "=============================================="
echo -e "${GREEN}L2TP/IPsec servers found: $FOUND${RESET}"
echo "=============================================="
}

    echo -e "${CYAN}└────────────────────────────────────────────┘${RESET}"

    echo -e "${GREEN}┌────────────────────────────────────────────┐${RESET}"
